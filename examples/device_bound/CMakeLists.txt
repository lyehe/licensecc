cmake_minimum_required(VERSION 3.21 FATAL_ERROR)
project(licensecc_device_bound_example LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LCC_PROJECT_NAME "test" CACHE STRING "Installed Licensecc project component")
find_package(licensecc CONFIG REQUIRED)
if(NOT LCC_ENABLE_DEVICE_IDENTITY OR
   (WIN32 AND NOT LCC_ENABLE_WINDOWS_TPM) OR
   (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND (NOT LCC_ENABLE_TPM2_OPENSSL OR NOT LCC_ENABLE_LINUX_DESKTOP)) OR
   (NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux"))
    message(FATAL_ERROR "This consumer requires an installed Windows or Linux package with device identity and its TPM provider enabled")
endif()

set(LCC_BOUND_FEATURE "DEFAULT" CACHE STRING "Licensed feature")
set(LCC_BOUND_DEVICE_LABEL "Workstation" CACHE STRING "Consent display label")
foreach(field APPLICATION_ID ENDPOINT_ORIGIN PORTAL_URL ISSUER LEASE_AUDIENCE PROOF_AUDIENCE PROJECT FEATURE CLIENT_ID DEVICE_LABEL)
    if(NOT DEFINED LCC_BOUND_${field} OR "${LCC_BOUND_${field}}" STREQUAL "")
        message(FATAL_ERROR "Set LCC_BOUND_${field} to the developer-owned application configuration")
    endif()
    set(value "${LCC_BOUND_${field}}")
    string(REPLACE "\\" "\\\\" value "${value}")
    string(REPLACE "\"" "\\\"" value "${value}")
    string(REPLACE "\n" "\\n" value "${value}")
    string(REPLACE "\r" "\\r" value "${value}")
    set(CONFIG_${field} "${value}")
endforeach()
if(NOT DEFINED LCC_BOUND_SIGNING_SPKI OR NOT EXISTS "${LCC_BOUND_SIGNING_SPKI}")
    message(FATAL_ERROR "Set LCC_BOUND_SIGNING_SPKI to the public RSA-3072 DER signing key")
endif()
set(LCC_BOUND_ADDITIONAL_SIGNING_SPKIS "" CACHE STRING "Additional public DER keys for signer overlap; semicolon-separated paths")
set(signing_paths "${LCC_BOUND_SIGNING_SPKI}" ${LCC_BOUND_ADDITIONAL_SIGNING_SPKIS})
list(LENGTH signing_paths signing_count)
if(signing_count GREATER 8)
    message(FATAL_ERROR "Configure at most eight public signing keys")
endif()
set(CONFIG_SIGNING_KEYS "")
set(signing_digests "")
foreach(signing_path IN LISTS signing_paths)
    if(NOT EXISTS "${signing_path}" OR IS_DIRECTORY "${signing_path}")
        message(FATAL_ERROR "Every public signing SPKI must be an existing DER file")
    endif()
    file(SIZE "${signing_path}" spki_size)
    if(spki_size LESS 1 OR spki_size GREATER 512)
        message(FATAL_ERROR "Each public signing SPKI must contain 1..512 DER bytes")
    endif()
    file(SHA256 "${signing_path}" spki_digest)
    if(spki_digest IN_LIST signing_digests)
        message(FATAL_ERROR "Duplicate public signing key")
    endif()
    list(APPEND signing_digests "${spki_digest}")
    file(READ "${signing_path}" spki_hex HEX)
    string(REGEX REPLACE "([0-9a-fA-F][0-9a-fA-F])" "0x\\1," spki_bytes "${spki_hex}")
    string(APPEND CONFIG_SIGNING_KEYS "{{${spki_bytes}}, ${spki_size}},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${signing_path}")
endforeach()
configure_file(configuration.hpp.in configuration.hpp @ONLY)
add_executable(licensecc_device_bound windows.cpp)
add_executable(licensecc_feature_sessions feature_sessions.cpp)
add_executable(licensecc_calculator calculator.cpp)
target_include_directories(licensecc_calculator PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(licensecc_calculator PRIVATE licensecc::licensecc_static)
target_include_directories(licensecc_feature_sessions PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(licensecc_feature_sessions PRIVATE licensecc::licensecc_static)
target_include_directories(licensecc_device_bound PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(licensecc_device_bound PRIVATE licensecc::licensecc_static)
if(MSVC)
    target_compile_options(licensecc_device_bound PRIVATE /utf-8)
    target_compile_options(licensecc_feature_sessions PRIVATE /utf-8)
    target_compile_options(licensecc_calculator PRIVATE /utf-8)
endif()
