cmake_minimum_required(VERSION 3.21 FATAL_ERROR)
project(licensecc_python_device_bound LANGUAGES CXX)
set(LCC_PROJECT_NAME "test" CACHE STRING "Installed Licensecc project component")
find_package(licensecc CONFIG REQUIRED)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT LCC_ENABLE_DEVICE_IDENTITY OR
   (WIN32 AND (NOT MSVC OR NOT LCC_ENABLE_WINDOWS_TPM)) OR
   (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND (NOT LCC_ENABLE_TPM2_OPENSSL OR NOT LCC_ENABLE_LINUX_DESKTOP)) OR
   (NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux"))
    message(FATAL_ERROR "The bridge requires an installed 64-bit Windows/MSVC or Linux TPM package")
endif()
add_library(licensecc_device_bound_bridge SHARED bridge.cpp)
if(WIN32)
    target_sources(licensecc_device_bound_bridge PRIVATE bridge.def)
else()
    # Export the same bounded ABI and force those objects out of the static archive.
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/bridge.def" bridge_exports REGEX "^[ ]+lcc_")
    set(export_map "{ global:\n")
    foreach(symbol IN LISTS bridge_exports)
        string(STRIP "${symbol}" symbol)
        string(APPEND export_map "  ${symbol};\n")
        target_link_options(licensecc_device_bound_bridge PRIVATE "LINKER:--undefined=${symbol}")
    endforeach()
    string(APPEND export_map "local: *; };\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bridge.map" "${export_map}")
    target_link_options(licensecc_device_bound_bridge PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/bridge.map")
endif()
target_compile_features(licensecc_device_bound_bridge PRIVATE cxx_std_17)
target_link_libraries(licensecc_device_bound_bridge PRIVATE licensecc::licensecc_static)
set_target_properties(licensecc_device_bound_bridge PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)
install(TARGETS licensecc_device_bound_bridge RUNTIME DESTINATION bin LIBRARY DESTINATION lib)

option(LCC_BRIDGE_BUILD_TESTS "Build a test-only bridge with the original exports" OFF)
if(LCC_BRIDGE_BUILD_TESTS AND WIN32)
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/bridge.def" old_exports)
    list(FILTER old_exports EXCLUDE REGEX "feature_session")
    list(JOIN old_exports "\n" old_definition)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/original-exports.def" "${old_definition}\n")
    add_library(licensecc_device_bound_original SHARED bridge.cpp "${CMAKE_CURRENT_BINARY_DIR}/original-exports.def")
    target_compile_features(licensecc_device_bound_original PRIVATE cxx_std_17)
    target_link_libraries(licensecc_device_bound_original PRIVATE licensecc::licensecc_static)
    set_target_properties(licensecc_device_bound_original PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)
endif()
